% Figure 1: MIST architecture, full-width sigconf version.
% Balanced layout: top row of four components, Inputs descriptor and
% Sniper Mutator and Outputs descriptor on the bottom row, SUT
% cluster on the right.  Dashed Jaeger->Oracle return arrow routes
% through the gap between the oracle and the SUT box.
\begin{tikzpicture}[
  font=\small,
  >={Stealth[length=2.4mm]},
  comp/.style    ={draw=blue!50!gray, line width=0.6pt, rounded corners=2.5pt,
                   fill=blue!4, minimum width=30mm, minimum height=17mm,
                   align=center, inner sep=2pt, text depth=0pt},
  sut/.style     ={draw=gray!55, line width=0.5pt, rounded corners=2pt,
                   fill=gray!10, minimum height=7mm, minimum width=24mm,
                   align=center, inner sep=2pt, font=\footnotesize},
  sutgap/.style  ={align=center, inner sep=1pt, font=\footnotesize\itshape,
                   text=gray!70},
  inputbox/.style={draw=gray!50, line width=0.5pt, rounded corners=2.5pt,
                   fill=gray!4, minimum height=17mm, minimum width=68mm,
                   align=left, inner sep=4pt, text depth=0pt,
                   font=\footnotesize},
  outlbl/.style  ={font=\footnotesize, align=center, anchor=center},
  flow/.style    ={->, thick, >=Stealth, draw=black!75},
  backflow/.style={->, thick, dashed, >=Stealth, draw=black!75}
]

% --- Top row: four pipeline components. ---
\node[comp] (ingest) at (0.0,0) {\textbf{(i) Spec Ingestor}\\{\scriptsize \textcolor{gray!50!black}{OpenAPI parser, emits config}}};
\node[comp] (registry) at (3.8,0) {\textbf{(ii) Semantic Dep. Registry}\\{\scriptsize \textcolor{gray!50!black}{ID-stem producer/consumer index}}};
\node[comp] (gen) at (7.6,0) {\textbf{(iii) Sequence Generator}\\{\scriptsize \textcolor{gray!50!black}{5-phase trace mining, Root API Mode}}};
\node[comp] (oracle) at (11.4,0) {\textbf{(v) Trace-Aware Oracle}\\{\scriptsize \textcolor{gray!50!black}{status + body + span + attribution}}};

% --- Bottom row: Inputs / Sniper / Outputs. ---
\node[inputbox] (inputs) at (1.9,-2.5) {\textbf{Inputs}\\\scriptsize\\\enspace$\bullet$ OpenAPI specification (one file or aggregated /v3/api-docs)\\\enspace$\bullet$ Jaeger / OpenTelemetry traces (one file or a directory)\\\enspace$\bullet$ run configuration (	exttt{.properties} file)};
\node[comp] (mut) at (7.6,-2.5) {\textbf{(iv) Sniper Mutator}\\{\scriptsize \textcolor{gray!50!black}{one fault per variant, round-robin}}};
\node[outlbl] (out) at (11.4,-2.5) {\textbf{Outputs}\\\scriptsize\\JUnit tests, Allure report,\\fault-detection report};

% --- System Under Test cluster (bigger items, more spacing). ---
\node[sut]    (gw)   at (15.0, 1.05)  {Gateway};
\node[sut]    (svc1) at (15.0, 0.40)  {ts-route};
\node[sut]    (svc2) at (15.0,-0.25)  {ts-order};
\node[sutgap] (dots) at (15.0,-0.70)  {$\vdots$};
\node[sut]    (jgr)  at (15.0,-1.20)  {Jaeger};
\begin{scope}[on background layer]
  \node[draw=gray!60, dotted, thick, rounded corners=4pt, inner sep=3mm,
        fit=(gw)(jgr),
        label={[font=\footnotesize\bfseries, gray!50!black]above:System Under Test}] {};
\end{scope}

% --- Wires ---
% Main horizontal pipeline
\draw[flow] (ingest)   -- (registry);
\draw[flow] (registry) -- (gen);
\draw[flow] (gen)      -- (oracle);

% Gen -> Sniper (vertical down)
\draw[flow] (gen.south) -- (mut.north);

% Sniper -> Oracle, right-angle path in the gap between rows
\draw[flow] (mut.north east) -| ([xshift=-3mm]oracle.south);

% Oracle -> Gateway (HTTP requests, solid arrow)
\draw[flow] ([yshift= 1.5mm]oracle.east) -- ([yshift= 1.5mm]oracle.east -| gw.west);

% Jaeger -> Oracle (dashed trace return).  Vertical segment routed
% in the gap between oracle.east (x=12.9) and SUT.west (~13.8) at
% x=13.4 (5mm left of jgr.west).
\draw[backflow] (jgr.west) -- ([xshift=-5mm]jgr.west) |- ([yshift=-1.5mm]oracle.east);

% Oracle -> Outputs label (vertical drop)
\draw[flow] (oracle.south) -- (out.north);

\end{tikzpicture}
