% Figure 1: MIST architecture.  Plain pipeline of five components +
% SUT cluster + run output.  Inputs (OpenAPI spec, Jaeger traces,
% .properties file) are NOT drawn on the canvas; they are named in the
% figure caption.  This keeps the figure free of crossing wires and
% matches the convention in recent tool-demo figures (RESTest @ ISSTA
% 2021, AutoRestTest @ ICSE 2025, ASTRAL @ ISSTA 2025).
\begin{tikzpicture}[
    font=\small,
    >={Stealth[length=2.4mm]},
    % All component boxes use a fixed height so the row is uniform.
    comp/.style={draw, rounded corners=2pt, fill=gray!8,
                 minimum height=12mm, minimum width=24mm,
                 align=center, inner sep=3pt, text depth=0pt},
    sut/.style ={draw, rounded corners=2pt, fill=blue!5,
                 minimum height=7mm, minimum width=18mm,
                 align=center, inner sep=2pt, font=\footnotesize},
    outlbl/.style={font=\footnotesize\itshape, align=center, anchor=north},
    flow/.style={->, thick, >=Stealth},
    backflow/.style={->, thick, dashed, >=Stealth}
]

% --- Four components on the main row (y=0); uniform two-line labels
%     so every box has the same height. ---
\node[comp] (ingest)   at ( 0.0, 0)   {(i) Spec\\Ingestor};
\node[comp] (registry) at ( 3.5, 0)   {(ii) Semantic\\Dep.\ Registry};
\node[comp] (gen)      at ( 7.0, 0)   {(iii) Sequence\\Generator};
\node[comp] (oracle)   at (11.0, 0)   {(v) Trace-Aware\\Oracle};

% --- Sniper Mutator directly below the Generator. ---
\node[comp] (mut)      at ( 7.0,-2.6) {(iv) Sniper\\Mutator};

% --- System Under Test cluster on the right.  Concrete service stubs
%     so the figure does not need \cdots; the two `ts-*' rows hint at
%     TrainTicket without naming any specific service. ---
\node[sut] (gw)   at (14.0, 1.1)  {Gateway};
\node[sut] (svc1) at (14.0, 0.25) {ts-route};
\node[sut] (svc2) at (14.0,-0.60) {ts-order};
\node[sut] (jgr)  at (14.0,-1.45) {Jaeger};
\begin{scope}[on background layer]
  \node[draw, dotted, thick, rounded corners=4pt, inner sep=3mm,
        fit=(gw)(jgr),
        label={[font=\footnotesize\bfseries]above:System Under Test}] {};
\end{scope}

% --- Run output drops below the oracle.  Anchored north so the label
%     grows downward and never collides with the mut->oracle wire that
%     approaches the oracle's south edge. ---
\node[outlbl] (out) at (11.0,-2.6)
  {JUnit tests, Allure report,\\fault-detection report};

% --- Main horizontal pipeline (left to right). ---
\draw[flow] (ingest)   -- (registry);
\draw[flow] (registry) -- (gen);
\draw[flow] (gen)      -- (oracle);

% --- Generator down to Sniper Mutator (clean vertical). ---
\draw[flow] (gen.south) -- (mut.north);

% --- Sniper Mutator back up into the Oracle, approaching from below.
%     Corner path: right at y=-2.6 then up at x just left of oracle.south,
%     so the rising segment does NOT collide with the output label that
%     sits directly below oracle.south. ---
\draw[flow] (mut.east) -| ([xshift=-4mm]oracle.south);

% --- Oracle <-> SUT: HTTP requests out (top), Jaeger trace back
%     (bottom).  Two parallel arrows with explicit +/- 1.5mm y-shift so
%     they remain visually distinct in print. ---
\draw[flow]     ([yshift= 1.5mm]oracle.east)
            -- ([yshift= 1.5mm]oracle.east -| gw.west);
\draw[backflow] (jgr.west)
            -- ([yshift=-1.5mm]oracle.east -| jgr.west)
            -- ([yshift=-1.5mm]oracle.east);

% --- Oracle to output (vertical, offset right of the mut->oracle wire
%     so the two do not overlap at the same x-coordinate). ---
\draw[flow] ([xshift=4mm]oracle.south) -- ([xshift=4mm]oracle.south |- out.north);

\end{tikzpicture}
